  documentName: "Slip_PBD_UASA_SK_Ranggu_2026",
  senderName: "Unit Pentaksiran SK Ranggu, Tawau",
  replyTo: "skranggu.tawau@moe.gov.my",
  emailSubject: "Slip Pelaporan PBD ((NAMA_MURID)) - SK Ranggu",
  safeBatchLimit: 25

function setupSheetStructure() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();
  const headers = [
    "BIL", "NO_MURID", "NAMA_MURID", "KELAS", "GURU_KELAS",
    "BM_M", "BM_G", "BM_TP", "BI_M", "BI_G", "BI_TP",
    "MT_M", "MT_G", "MT_TP", "SN_M", "SN_G", "SN_TP",
    "SEJ_M", "SEJ_G", "SEJ_TP", "PI_M", "PI_G", "PI_TP",
    "TOTAL", "PURATA", "POSISI", "HADIR", "ULASAN", "EMAIL_PENJAGA",
    "STATUS_JANA", "TARIKH_JANA", "CATATAN_JANA", "LINK_PDF"
  ];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
}
