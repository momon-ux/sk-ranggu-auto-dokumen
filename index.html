const DEFAULT_CONFIG = {
  documentName: "Slip Peperiksaan Akhir SK Ranggu 2026",
  senderName: "Unit Peperiksaan & Pentaksiran SK Ranggu, Tawau",

  emailSubject: "Slip Keputusan Pentaksiran ((NAMA_MURID)) - SK Ranggu Tawau",
  safeBatchLimit: 25
};

function doGet(e) {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle("Sistem Auto Dokumen SK Ranggu")
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function setupSheetStructure() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Data_Peperiksaan") || ss.getActiveSheet();
  sheet.setName("Data_Peperiksaan");
  
  const headers = [
    "BIL", "NO_MURID", "NAMA_MURID", "KELAS", "SESI",
    "BM_M", "BM_G", "BI_M", "BI_G", "MT_M", "MT_G", "SN_M", "SN_G", "SEJ_M", "SEJ_G", "PI_M", "PI_G",
    "TOTAL", "PURATA", "POSISI", "HADIR", "ULASAN", "EMAIL_PENJAGA",
    "STATUS_JANA", "TARIKH_JANA", "CATATAN_JANA", "LINK_PDF"
  ];
  
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length)
    .setBackground("#0F172A")
    .setFontColor("#FFFFFF")
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
  sheet.setFrozenRows(1);
  return { success: true, message: "Struktur sheet berjaya disediakan!" };
}

function janaSlipsBatch(options) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Data_Peperiksaan") || ss.getActiveSheet();
  const allData = sheet.getDataRange().getValues();
  const headers = allData[0];
  
  const colMap = {};
  headers.forEach((h, idx) => colMap[h] = idx + 1);
  
  const targetFolder = options.targetFolderId 
    ? DriveApp.getFolderById(options.targetFolderId) 
    : DriveApp.createFolder("PDF_Slip_Peperiksaan_SK_Ranggu");
    
  const templateFile = DriveApp.getFileById(options.templateDocId);
  
  let countSuccess = 0;
  for (let r = 1; r < allData.length; r++) {
    const row = allData[r];
    const studentObj = {};
    headers.forEach((h, idx) => studentObj[h] = row[idx]);
    
    const tempDoc = templateFile.makeCopy(`TEMP_${studentObj.NAMA_MURID}`, targetFolder);
    const doc = DocumentApp.openById(tempDoc.getId());
    const body = doc.getBody();
    
    headers.forEach(h => {
      body.replaceText(`<<${h}>>`, String(studentObj[h] || ""));
      body.replaceText(`\\(\\(${h}\\)\\)`, String(studentObj[h] || ""));
    });
    doc.saveAndClose();
    
    const pdfBlob = tempDoc.getAs(MimeType.PDF);
    pdfBlob.setName(`Slip_${studentObj.NAMA_MURID}_${studentObj.NO_MURID}.pdf`);
    const pdfFile = targetFolder.createFile(pdfBlob);
    tempDoc.setTrashed(true);
    
    if (options.mode === "PDF_EMAIL" && studentObj.EMAIL_PENJAGA) {
      MailApp.sendEmail({
        to: studentObj.EMAIL_PENJAGA,
        subject: `Slip Keputusan Pentaksiran ${studentObj.NAMA_MURID}`,
        body: `Assalamualaikum,\n\nBerikut dilampirkan slip keputusan bagi ${studentObj.NAMA_MURID} (${studentObj.KELAS}).`,
        attachments: [pdfFile.getAs(MimeType.PDF)]
      });
    }
    
    sheet.getRange(r + 1, colMap["STATUS_JANA"]).setValue("BERJAYA");
    countSuccess++;
  }
  return { success: true, processed: countSuccess };
}
